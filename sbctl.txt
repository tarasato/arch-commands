sudo pacman -Syu sbctl
sbctl status
sbctl create-keys
sbctl enroll-keys -m -f

sudo grub-install \
  --target=x86_64-efi \
  --efi-directory=/efi \
  --bootloader-id=GRUB \
  --modules="tpm" \
  --disable-shim-lock

sudo sbctl verify
sudo sbctl sign -s /path/to/grubx64.efi
sudo sbctl sign -s /boot/vmlinuz-linux

sudo grub-mkconfig -o /boot/grub/grub.cfg
